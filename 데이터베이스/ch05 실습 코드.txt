-- ---------------------------------------------------------------------
-- [p.10] ERD (참고 — 이 실습에서 만들지는 않음)
--   교수는 과목을 강의한다 / 교수는 학생을 상담한다
--   학생은 교수에게 상담을 받는다 / 학생은 과목을 수강한다
--   네모 = 개체(교수, 학생, 과목), 마름모 = 관계(상담, 강의, 수강)
--   p.11 스키마는 이 중 학생·교수·과목과 "강의"·"상담" 관계를 테이블로 만든다 (수강은 없음).
-- ---------------------------------------------------------------------


-- ---------------------------------------------------------------------
-- [p.11] 테이블 스키마 지시문
--   1) 학생 테이블은 학생아이디, 학생이름, 학과, 이수학기, 장학금 속성으로
--      구성되며 학생아이디가 primary key
--   2) 교수 테이블은 교수아이디, 교수이름, 소속, 연락처, 전공학위 속성으로
--      구성되며 교수아이디가 primary key
--   3) 과목 테이블은 과목아이디, 과목명, 과목학점, 과목유형, 개설학과 속성으로
--      구성되며 과목아이디가 primary key
--   4) 강의 테이블은 강의아이디, 강사, 과목, 강의실, 강의교시, 강의요일 속성으로
--      구성되며 강의아이디가 primary key
--      강사는 교수 테이블에서 교수아이디를 참조하는 외래키로 구성되고,
--      과목은 과목 테이블에서 과목아이디를 참조하는 외래키로 구성된다
--   5) 상담 테이블은 상담아이디, 상담학생, 상담교수, 상담실, 상담일시, 상담내용
--      속성으로 구성되며 상담아이디가 PRIMARY KEY          (09-30 추가된 조건)
--      + 상담학생은 학생 테이블의 학생아이디, 상담교수는 교수 테이블의 교수아이디를
--        참조하는 외래키 (지시문엔 없지만 4번 강의와 같은 원리로 추가)
--
--   만드는 순서: 참조당하는 테이블(학생·교수·과목)을 먼저 → 강의·상담을 나중에
--   (강의·상담을 먼저 만들면 참조할 테이블이 없어서 ERROR 1824)
-- ---------------------------------------------------------------------

DROP DATABASE IF EXISTS 학사;
CREATE DATABASE 학사;
USE 학사;

CREATE TABLE 학생 (
    학생아이디  VARCHAR(20) NOT NULL,
    학생이름    VARCHAR(20) NOT NULL,
    학과        VARCHAR(30),
    이수학기    INT,
    장학금      INT,
    PRIMARY KEY (학생아이디)
);

CREATE TABLE 교수 (
    교수아이디  VARCHAR(20) NOT NULL,
    교수이름    VARCHAR(20) NOT NULL,
    소속        VARCHAR(30),
    연락처      VARCHAR(20),
    전공학위    VARCHAR(30),
    PRIMARY KEY (교수아이디)
);

CREATE TABLE 과목 (
    과목아이디  VARCHAR(20) NOT NULL,
    과목명      VARCHAR(30) NOT NULL,
    과목학점    INT,
    과목유형    VARCHAR(10),
    개설학과    VARCHAR(30),
    PRIMARY KEY (과목아이디)
);

CREATE TABLE 강의 (
    강의아이디  VARCHAR(20) NOT NULL,
    강사        VARCHAR(20),
    과목        VARCHAR(20),
    강의실      VARCHAR(20),
    강의교시    INT,
    강의요일    VARCHAR(5),
    PRIMARY KEY (강의아이디),
    FOREIGN KEY (강사) REFERENCES 교수(교수아이디),
    FOREIGN KEY (과목) REFERENCES 과목(과목아이디)
);

CREATE TABLE 상담 (
    상담아이디  VARCHAR(20) NOT NULL,
    상담학생    VARCHAR(20),
    상담교수    VARCHAR(20),
    상담실      VARCHAR(20),
    상담일시    DATETIME,
    상담내용    VARCHAR(100),
    PRIMARY KEY (상담아이디),
    FOREIGN KEY (상담학생) REFERENCES 학생(학생아이디),
    FOREIGN KEY (상담교수) REFERENCES 교수(교수아이디)
);

SHOW TABLES;
DESC 강의;
--   강사·과목의 Key 칸이 MUL = 외래키(인덱스)가 걸린 열

-- 강의 시간표 — 강의의 강사·과목 아이디를 교수이름·과목명으로 바꿔 보기 (3개 테이블 JOIN)
SELECT l.강의아이디, c.과목명, p.교수이름, l.강의실, l.강의요일, l.강의교시
FROM 강의 l
JOIN 교수 p ON l.강사 = p.교수아이디
JOIN 과목 c ON l.과목 = c.과목아이디
ORDER BY l.강의아이디;
--   → lc01 데이터베이스 김교수 G101 수 2 / lc02 인공지능개론 이교수 G203 월 3 /
--     lc03 알고리즘 박교수 H305 화 1 / lc04 빅데이터분석 이교수 G203 목 5

-- 이교수가 맡은 강의 (JOIN + WHERE)
SELECT p.교수이름, c.과목명, l.강의요일, l.강의교시
FROM 강의 l
JOIN 교수 p ON l.강사 = p.교수아이디
JOIN 과목 c ON l.과목 = c.과목아이디
WHERE p.교수이름 = '이교수';
--   → 이교수 인공지능개론 월 3 / 이교수 빅데이터분석 목 5

-- 교수별 담당 강의 수 — 강의 없는 교수도 보이게 LEFT JOIN
SELECT p.교수이름, COUNT(l.강의아이디) AS 강의수
FROM 교수 p
LEFT JOIN 강의 l ON p.교수아이디 = l.강사
GROUP BY p.교수아이디, p.교수이름;
--   → 김교수 1 / 이교수 2 / 박교수 1 / 최교수 0
--     (INNER JOIN이면 최교수가 아예 안 나옴. COUNT(l.강의아이디)는 NULL을 안 세서 0)

-- 강의를 맡지 않은 교수 — LEFT JOIN 후 짝 없는(NULL) 행만
SELECT p.교수이름
FROM 교수 p
LEFT JOIN 강의 l ON p.교수아이디 = l.강사
WHERE l.강의아이디 IS NULL;
--   → 최교수

-- 휴먼AI공학 개설 과목의 총 강의 학점 (JOIN + WHERE + 집계)
SELECT SUM(c.과목학점) AS 총학점
FROM 강의 l
JOIN 과목 c ON l.과목 = c.과목아이디
WHERE c.개설학과 = '휴먼AI공학';
--   → 8 (데이터베이스 3 + 인공지능개론 3 + 빅데이터분석 2)

-- 학생 테이블 단독 조회
SELECT 학과, COUNT(*) AS 학생수, AVG(장학금) AS 평균장학금
FROM 학생
GROUP BY 학과;
--   → 휴먼AI공학 3 266666.6667 / 컴퓨터공학 2 1000000.0000
--     (최수아 장학금 NULL은 AVG에서 빠짐 → 컴퓨터공학 평균 = 1000000/1)


-- ---------------------------------------------------------------------
-- 상담 테이블 조회 (5번 추가 조건)
--   JOIN 조건 = 외래키 열 = 참조하는 기본키 (상담학생 = 학생아이디, 상담교수 = 교수아이디)
-- ---------------------------------------------------------------------

-- 상담 기록을 학생이름·교수이름으로 (상담 — 학생 — 교수 3개 테이블 JOIN)
SELECT cs.상담아이디, s.학생이름, p.교수이름, cs.상담실, cs.상담일시, cs.상담내용
FROM 상담 cs
JOIN 학생 s ON cs.상담학생 = s.학생아이디
JOIN 교수 p ON cs.상담교수 = p.교수아이디
ORDER BY cs.상담일시;
--   → 5행: cs01 김민준 김교수 … / cs02 이서연 김교수 … / cs03 김민준 이교수 … /
--          cs04 박지훈 박교수 … / cs05 최수아 김교수 …

-- 김민준이 상담받은 교수 (JOIN + WHERE)
SELECT s.학생이름, p.교수이름, cs.상담일시, cs.상담내용
FROM 상담 cs
JOIN 학생 s ON cs.상담학생 = s.학생아이디
JOIN 교수 p ON cs.상담교수 = p.교수아이디
WHERE s.학생이름 = '김민준';
--   → 김민준 김교수 대학원 진학 상담 / 김민준 이교수 인턴십 지원

-- 교수별 상담 건수 — 상담 없는 교수도 보이게 LEFT JOIN
SELECT p.교수이름, COUNT(cs.상담아이디) AS 상담건수
FROM 교수 p
LEFT JOIN 상담 cs ON p.교수아이디 = cs.상담교수
GROUP BY p.교수아이디, p.교수이름
ORDER BY 상담건수 DESC;
--   → 김교수 3 / 이교수 1 / 박교수 1 / 최교수 0

-- 상담을 한 번도 받지 않은 학생
SELECT s.학생이름
FROM 학생 s
LEFT JOIN 상담 cs ON s.학생아이디 = cs.상담학생
WHERE cs.상담아이디 IS NULL;
--   → 정우진

-- 상담실별 상담 건수 (상담 테이블 단독 GROUP BY)
SELECT 상담실, COUNT(*) AS 상담건수
FROM 상담
GROUP BY 상담실;
--   → G501 3 / G502 1 / H410 1

-- 학생테이블에서 장학금금액이 100만원넘는 학생의이름소속 이수학기 장학금을 조회

-- 장학금을 가장많이받은 학생 이름 학과 장학금 세명만 장학금 내림차순으
--
